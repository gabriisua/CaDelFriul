    Client component with:

    - User info section at top: avatar placeholder circle + "Guest User" name (placeholder)
    - Nav links: Overview (/dashboard), Profile (/dashboard/profile), Addresses (/dashboard/addresses), Orders (/dashboard/orders), Reservations (/dashboard/reservations)

    ```tsx
    "use client";

    import Link from "next/link";
    import { usePathname } from "next/navigation";
    import { Menu, X } from "lucide-react";
    import { useState } from "react";
    import { Button } from "@/components/ui/button";
    import { Separator } from "@/components/ui/separator";
    import { cn } from "@/lib/utils";

    const navItems = [
      { href: "/dashboard", label: "Overview" },
      { href: "/dashboard/profile", label: "Profile" },
      { href: "/dashboard/addresses", label: "Addresses" },
      { href: "/dashboard/orders", label: "Orders" },
      { href: "/dashboard/reservations", label: "Reservations" },
    ];

    export default function SidebarNav() {
      const [mobileOpen, setMobileOpen] = useState(false);
      const pathname = usePathname();

      const sidebarContent = (
        <div className="flex h-full flex-col">
          {/* User info */}
          <div className="flex items-center gap-3 px-6 py-6">
            <div className="flex size-10 items-center justify-center rounded-full bg-accent text-sm font-medium text-accent-foreground">
              G
            </div>
            <div>
              <p className="text-sm font-medium text-sidebar-foreground">Guest User</p>
              <p className="text-xs text-muted-foreground">guest@example.com</p>
            </div>
          </div>

          <Separator />

          {/* Nav links */}
          <nav className="flex-1 space-y-1 px-3 py-4">
            {navItems.map((item) => {
              const isActive =
                item.href === "/dashboard"
                  ? pathname === "/dashboard"
                  : pathname.startsWith(item.href);
              return (
                <Link
                  key={item.href}
                  href={item.href}
                  onClick={() => setMobileOpen(false)}
                  className={cn(
                    "flex items-center rounded-lg px-3 py-2 text-sm font-medium transition-colors",
                    isActive
                      ? "bg-accent/10 text-accent border-l-2 border-accent"
                      : "text-sidebar-foreground/70 hover:bg-sidebar-accent hover:text-sidebar-foreground"
                  )}
                >
                  {item.label}
                </Link>
              );
            })}
          </nav>

          {/* Logout */}
          <div className="px-3 pb-6">
            <Separator className="mb-4" />
            <Button
              variant="ghost"
              className="w-full justify-start text-muted-foreground"
              disabled
            >
              Logout
            </Button>
          </div>
        </div>
      );

      return (
        <>
          {/* Mobile toggle */}
          <button
            className="fixed left-4 top-4 z-50 md:hidden"
            onClick={() => setMobileOpen(!mobileOpen)}
            aria-label={mobileOpen ? "Close sidebar" : "Open sidebar"}
          >
            {mobileOpen ? (
              <X className="size-5 text-foreground" />
            ) : (
              <Menu className="size-5 text-foreground" />
            )}
          </button>

          {/* Desktop sidebar */}
          <aside className="hidden h-full w-64 shrink-0 border-r border-border bg-sidebar md:block">
            {sidebarContent}
          </aside>

          {/* Mobile sidebar overlay */}
          {mobileOpen && (
            <div className="fixed inset-0 z-40 md:hidden">
              <div
                className="absolute inset-0 bg-black/50"
                onClick={() => setMobileOpen(false)}
              />
              <aside className="relative h-full w-64 bg-sidebar shadow-lg">
                {sidebarContent}
              </aside>
            </div>
          )}
        </>
      );
    }
    ```
