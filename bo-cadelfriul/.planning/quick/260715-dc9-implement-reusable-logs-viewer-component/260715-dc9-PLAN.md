   ```typescript
   openLogs(staffMember: Staff): void {
     this.staffService.getStaffLogs(staffMember.id).subscribe({
       next: (logs) => {
         this.selectedLogs = logs;
         this.logsTitle = 'Logs - ' + staffMember.firstName + ' ' + staffMember.lastName;
         this.showLogsDialog = true;
         this.cdr.detectChanges();
       },
       error: (err) => {
         console.error('Failed to load staff logs:', err);
         this.error = 'Failed to load staff logs.';
         this.cdr.detectChanges();
       },
     });
   }
   ```
